git init # иницилизирует проекта

git add <путь> # индексирует файлы по пути
git add . # индексация всего

git commit -m "Название коммита" # Создаёт сохранение, в которое попадают все проиндексированные файл

git restore <путь> # Откатывает все непроиндексированные изменения до состояния текущего коммита

git diff # Показывает непроиндексированные изменения
git diff --staged # Показывает проиндексированные изменения
